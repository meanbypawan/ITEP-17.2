Creating Database :-
  use empdb
  
Creating Collection
  db.createCollection("employee")
Creating  capped collection
  db.createCollection("employee",{capped:true,size:1000, max:5})
Checking whether it is capped or uncapped
  db.employee.isCapped()
  
Drop Collection
  db.employee.drop()
  
Inserting single document
  db.employee.insertOne({"name":"Khusi"})
Inserting multiple record
  db.employee.insertMany([{},{},{}])
  
Fetching all record
  db.employee.find()
  
Using where condition
  db.employee.find({skill: "Java"})
Using projection
  db.employee.find({},{name:1, skill: 1, salary: 1, _id:0})

Limiting record
  db.employee.find().limit(5)

Skiping record
  db.employee.find().skip(2).limit(5) :=> Skip first 2 then fetch next 5
Counting
  db.employee.find().count()
--------------------------------------------
  $gt, $gte, $lt, $lte, $eq, $ne
  $and: [{},{}]
  select * from employee where salary >=50000
  db.employee.find({
    salary: {$gt:50000}
  })
  
  select * from employee where salary >=6000 and salary <=70000
  
  db.employee.find({
    $and:[{salary: {$gte:60000}}, {salary: {$lte:70000}}]
  })
  
  select * from employee where skill="Java" or skill="Python"
  db.employee.find({
    $or:[{skill: "Java"},{skill: "Python"}]
  })
  
  db.employee.find({salary: {$ne:70000}})

  select max(salary) from employee
  
Sorting
  db.employee.find().sort({salary:1}) ---> Ascending
  or
  db.employee.find().sort({salary: -1}) --> Descending
